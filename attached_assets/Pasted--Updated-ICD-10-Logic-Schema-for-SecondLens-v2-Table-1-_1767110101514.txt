-- ✅ Updated ICD-10 Logic Schema for SecondLens v2

-- Table 1: Clinical Term Mappings
CREATE TABLE icd_term_map (
    id SERIAL PRIMARY KEY,
    term TEXT NOT NULL,
    normalized_term TEXT,
    matched_code VARCHAR(10),
    match_confidence INT DEFAULT 100,
    combo_trigger BOOLEAN DEFAULT FALSE,
    requires_secondary_code BOOLEAN DEFAULT FALSE
);

-- Table 2: Logic Rules (now includes suppression and reason-not-selected fields)
CREATE TABLE icd_logic_rules (
    id SERIAL PRIMARY KEY,
    condition_cluster TEXT,
    trigger_terms TEXT[],
    primary_code VARCHAR(10),
    secondary_code VARCHAR(10),
    rule_description TEXT,
    suppression_keywords TEXT[],
    reason_not_selected TEXT
);

-- Table 3: Explanation Templates (shown when code is selected)
CREATE TABLE icd_explanations (
    id SERIAL PRIMARY KEY,
    icd_code VARCHAR(10),
    explanation TEXT
);

-- Table 4: Related Code Explanations (shown when code is excluded)
CREATE TABLE icd_related_exclusions (
    id SERIAL PRIMARY KEY,
    icd_code VARCHAR(10),
    explanation TEXT
);
